import pandas as pd
# 读取云栖茶饮的订单表
df = pd.read_excel("orders.xlsx")
# 先看看数据长什么样
print(df.info())
print(df.head())
# 处理缺失值：金额列填充 0，门店列用「未知」填充
df["金额"] = df["金额"].fillna(0)
df["门店"] = df["门店"].fillna("未知")
# 筛选出有效订单（金额 > 0 且 状态为「已完成」）
valid_orders = df[(df["金额"] > 0) & (df["状态"] == "已完成")]
# 按门店分组汇总销售额
store_sales = valid_orders.groupby("门店")["金额"].sum().reset_index()
store_sales = store_sales.sort_values("金额", ascending=False)
print(store_sales)
